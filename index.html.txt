                <span>總合結果：</span>

    <script>
        const rollBtn = document.getElementById('rollBtn');
        const diceTypeSelect = document.getElementById('diceType');
        const diceCountInput = document.getElementById('diceCount');
        const diceContainer = document.getElementById('diceContainer');
        const totalResult = document.getElementById('totalResult');

        rollBtn.addEventListener('click', () => {
            const type = parseInt(diceTypeSelect.value);
            const count = parseInt(diceCountInput.value) || 1;
            let sum = 0;
            diceContainer.innerHTML = '';
            for(let i=0; i<count; i++) {
                const val = Math.floor(Math.random() * type) + 1;
                sum += val;
                const dieEl = document.createElement('div');
                dieEl.className = 'die';
                dieEl.textContent = val;
                diceContainer.appendChild(dieEl);
            }
            totalResult.textContent = sum;
        });
    </script>
